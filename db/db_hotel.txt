CREATE DATABASE Hotel;
USE Hotel;


-- Table Entreprise
CREATE TABLE Entreprise (
    ID_Entreprise INT AUTO_INCREMENT PRIMARY KEY,
    Nom_Entreprise VARCHAR(60) NOT NULL,
    Logo VARCHAR(255) NOT NULL,
    Email_Entreprise VARCHAR(40) NOT NULL
);

-- Table Hotel
CREATE TABLE Hotel (
    ID_Hotel INT AUTO_INCREMENT PRIMARY KEY,
    ID_Entreprise INT,
    Nom_Hotel VARCHAR(60) NOT NULL,
    Pays VARCHAR(50),
    Ville VARCHAR(50),
    Adresse_Hotel VARCHAR(255),
    Nbr_Chambre INT CHECK (Nbr_Chambre >= 0),
    Nbr_Etoile INT CHECK (Nbr_Etoile BETWEEN 1 AND 5),
    Type_Hotel ENUM('Business', 'Luxe', 'Standard', 'Familiale','Maisons'),
    Latitude DOUBLE NOT NULL,
    Longitude DOUBLE NOT NULL,
    FOREIGN KEY (ID_Entreprise) REFERENCES Entreprise(ID_Entreprise) ON DELETE CASCADE
);

-- Table Chambre
CREATE TABLE Chambre (
    ID_Chambre INT AUTO_INCREMENT PRIMARY KEY,
    ID_Hotel INT NOT NULL,
    Num_Chambre INT NOT NULL,
    Type ENUM('Standard', 'Familiale', 'Suite', 'Deluxe', 'VIP') NOT NULL,
    Prix DECIMAL(10,2) CHECK (Prix >= 0),
    Statut_Chambre ENUM('Libre', 'Occupée', 'Réservée', 'En nettoyage', 'En maintenance') NOT NULL,
    Description_Chambre TEXT,
    Type_Offre ENUM('Promotion', 'Speciale','Aucun') NOT NULL,
    Speciale TEXT,
    Promotion DECIMAL(20,0),
    FOREIGN KEY (ID_Hotel) REFERENCES Hotel(ID_Hotel) ON DELETE CASCADE
);

-- Table Image pour Chambre
CREATE TABLE Image_C (
    ID_Image INT AUTO_INCREMENT PRIMARY KEY,
    ID_Chambre INT NOT NULL,
    URL VARCHAR(255) NOT NULL,
    FOREIGN KEY (ID_Chambre) REFERENCES Chambre(ID_Chambre) ON DELETE CASCADE
);

-- Table Options_Chambre (anciennement Option)
CREATE TABLE Options_Chambre (
    ID_Option INT AUTO_INCREMENT PRIMARY KEY,
    ID_Chambre INT NOT NULL,
    Option_Chambre ENUM('Espace extérieur', 'Vue sur l\'océan', 'Vue sur la ville', 'Terrasse', 'Climatisation', 'Chauffage', 'Wi-Fi inclus', 'Cuisine') NOT NULL,
    FOREIGN KEY (ID_Chambre) REFERENCES Chambre(ID_Chambre) ON DELETE CASCADE
);



-- Table Compte (pour gérer les comptes clients)
CREATE TABLE Compte (
    ID_Compte INT AUTO_INCREMENT PRIMARY KEY,
    Username VARCHAR(255) NOT NULL,
    Email VARCHAR(100) UNIQUE,
    Password VARCHAR(255) NOT NULL,
    Type_Compte ENUM('Client', 'Personnel', 'Admin', 'Manager') NOT NULL
);

-- Table Client
CREATE TABLE Client (
    ID_Client INT AUTO_INCREMENT PRIMARY KEY,
    ID_Compte INT NOT NULL,
    Nom_Client VARCHAR(30) ,
    Prenom_Client VARCHAR(30) ,
    DD_Naissance DATE,
    Telephone VARCHAR(10),
    Adresse_Client VARCHAR(255),
    FOREIGN KEY (ID_Compte) REFERENCES Compte(ID_Compte) ON DELETE CASCADE
);

-- Table Reservation
CREATE TABLE Reservation (
    ID_Reservation INT AUTO_INCREMENT PRIMARY KEY,
    ID_Chambre INT NOT NULL,
    ID_Client INT NOT NULL,
    Date_Debut DATE NOT NULL,
    Date_Fin DATE NOT NULL,
    Statut_Reservation ENUM('En cours', 'Confirmée', 'Annulée') NOT NULL,
    FOREIGN KEY (ID_Chambre) REFERENCES Chambre(ID_Chambre) ON DELETE CASCADE,
    FOREIGN KEY (ID_Client) REFERENCES Client(ID_Client) ON DELETE CASCADE
);

-- Table Avis
CREATE TABLE Avis (
    ID_Avis INT AUTO_INCREMENT PRIMARY KEY,
    ID_Chambre INT NOT NULL,
    ID_Client INT NOT NULL,
    Nb_etoile INT CHECK (Nb_etoile BETWEEN 1 AND 10),
    Description_Avis TEXT,
    Date_Avis DATETIME ,
    FOREIGN KEY (ID_Chambre) REFERENCES Chambre(ID_Chambre) ON DELETE CASCADE,
    FOREIGN KEY (ID_Client) REFERENCES Client(ID_Client) ON DELETE CASCADE
);

-- Table Personnel
CREATE TABLE Personnel (
    ID_Personnel INT AUTO_INCREMENT PRIMARY KEY,
    ID_Hotel INT NOT NULL,
    ID_Compte INT,
    Nom_Personnel VARCHAR(30) NOT NULL,
    Prenom_Personnel VARCHAR(30) NOT NULL,
    DD_Naissance DATE,
    Tache VARCHAR(30),
    Salaire INT,
    FOREIGN KEY (ID_Hotel) REFERENCES Hotel(ID_Hotel) ON DELETE CASCADE,
    FOREIGN KEY (ID_Compte) REFERENCES Compte(ID_Compte)
);


-- Table Services
CREATE TABLE Services (
    ID_Service INT AUTO_INCREMENT PRIMARY KEY,
    ID_Hotel INT NOT NULL,
    Nom_Service ENUM('Restaurant', 'Salle de Sport', 'Piscine', 'Parking', 'Massages') NOT NULL,
    Description_Service TEXT,
    Prix DECIMAL(10,2) CHECK (Prix >= 0),
    FOREIGN KEY (ID_Hotel) REFERENCES Hotel(ID_Hotel) ON DELETE CASCADE
);



-- Table Image pour Services
CREATE TABLE Image_S (
    ID_Image INT AUTO_INCREMENT PRIMARY KEY,
    ID_Service INT NOT NULL,
    URL VARCHAR(255) NOT NULL,
    FOREIGN KEY (ID_Service) REFERENCES Services(ID_Service) ON DELETE CASCADE
);

-- Table Image pour Hotel
CREATE TABLE Image_H (
    ID_Image INT AUTO_INCREMENT PRIMARY KEY,
    ID_Hotel INT NOT NULL,
    URL VARCHAR(255) NOT NULL,
    FOREIGN KEY (ID_Hotel) REFERENCES Hotel(ID_Hotel) ON DELETE CASCADE
);
